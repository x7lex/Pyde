from utils import greet

def fibonacci(n: int) -> int:
    """Return the n-th Fibonacci number."""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

if __name__ == "__main__":
    print(greet("Pyde"))
    print([fibonacci(i) for i in range(10)])
